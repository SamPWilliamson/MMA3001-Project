 29class NumberedCanvas(canvas.Canvas):
 30    """Two-pass canvas to dynamically compute and display total page count."""
 31
 32    def __init__(self, *args, **kwargs):
 33        super().__init__(*args, **kwargs)
 34        self._saved_page_states = []
 35
 36    def showPage(self):
 37        self._saved_page_states.append(dict(self.__dict__))
 38        self._startPage()
 39
 40    def save(self):
 41        num_pages = len(self._saved_page_states)
 42        for state in self._saved_page_states:
 43            self.__dict__.update(state)
 44            self.draw_page_decorations(num_pages)
 45            super().showPage()
 46        super().save()
 47
 48    def draw_page_decorations(self, page_count: int):
 49        self.saveState()
 50        self.setFont("Helvetica-Bold", 8)
 51        self.setFillColor(colors.HexColor("#64748B"))
 52
 53        # Header (pages 2+)
 54        if self._pageNumber > 1:
 55            self.setStrokeColor(colors.HexColor("#CBD5E1"))
 56            self.setLineWidth(0.6)
 57            self.line(40, 808, 555, 808)
 58            self.drawString(40, 812, "MMA3001 MAJOR PROJECT: PLAIN-ENGLISH UNDERSTANDING GUIDE")
 59            self.setFont("Helvetica", 8)
 60            self.drawRightString(555, 812, "Monash Building 69, Room G.57")
 61
 62        # Footer (all pages)
 63        self.setStrokeColor(colors.HexColor("#CBD5E1"))

 66        self.setFont("Helvetica", 8)

 69        self.drawRightString(555, 30, page_str)
